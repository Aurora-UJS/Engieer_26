void Kalman_Filter_xhatMinusUpdate(KalmanFilter_t *kf)
{
if (!kf->SkipEq1) // 如果未跳过此步骤
{
if (kf->uSize > 0) // 如果存在控制输入
{
kf->temp_vector.numRows = kf->xhatSize; // 设置临时矩阵的行数
kf->temp_vector.numCols = 1; // 设置临时矩阵的列数
kf->MatStatus = Matrix_Multiply(&kf->F, &kf->xhat, &kf->temp_vector); // 计算F·xhat(k-1)
kf->temp_vector1.numRows = kf->xhatSize; // 设置另一个临时矩阵的行数
kf->temp_vector1.numCols = 1; // 设置另一个临时矩阵的列数
kf->MatStatus = Matrix_Multiply(&kf->B, &kf->u, &kf->temp_vector1); // 计算B·u
kf->MatStatus = Matrix_Add(&kf->temp_vector, &kf->temp_vector1, &kf->xhatminus); // 计算F·xhat(k-1) + B·u
}
else
{
kf->MatStatus = Matrix_Multiply(&kf->F, &kf->xhat, &kf->xhatminus); // 仅计算F·xhat(k-1)
}
}
}
void Kalman_Filter_PminusUpdate(KalmanFilter_t *kf)
{
if (!kf->SkipEq2) // 如果未跳过此步骤
{
kf->MatStatus = Matrix_Transpose(&kf->F, &kf->FT); // 计算F的转置矩阵FT
kf->MatStatus = Matrix_Multiply(&kf->F, &kf->P, &kf->Pminus); // 计算F·P(k-1)
kf->temp_matrix.numRows = kf->Pminus.numRows; // 设置临时矩阵的行数
kf->temp_matrix.numCols = kf->FT.numCols; // 设置临时矩阵的列数
kf->MatStatus = Matrix_Multiply(&kf->Pminus, &kf->FT, &kf->temp_matrix); // 计算F·P(k-1)·FT
kf->MatStatus = Matrix_Add(&kf->temp_matrix, &kf->Q, &kf->Pminus); // 计算F·P(k-1)·FT + Q
}
}
void Kalman_Filter_SetK(KalmanFilter_t *kf)
{
if (!kf->SkipEq3) // 如果未跳过此步骤
{
kf->MatStatus = Matrix_Transpose(&kf->H, &kf->HT); // 计算H的转置矩阵HT
kf->temp_matrix.numRows = kf->H.numRows; // 设置临时矩阵的行数
kf->temp_matrix.numCols = kf->Pminus.numCols; // 设置临时矩阵的列数
kf->MatStatus = Matrix_Multiply(&kf->H, &kf->Pminus, &kf->temp_matrix); // 计算H·P'(k)
kf->temp_matrix1.numRows = kf->temp_matrix.numRows; // 设置另一个临时矩阵的行数
kf->temp_matrix1.numCols = kf->HT.numCols; // 设置另一个临时矩阵的列数
kf->MatStatus = Matrix_Multiply(&kf->temp_matrix, &kf->HT, &kf->temp_matrix1); // 计算H·P'(k)·HT
kf->S.numRows = kf->R.numRows; // 设置S矩阵的行数
kf->S.numCols = kf->R.numCols; // 设置S矩阵的列数
kf->MatStatus = Matrix_Add(&kf->temp_matrix1, &kf->R, &kf->S); // 计算H·P'(k)·HT + R
kf->MatStatus = Matrix_Inverse(&kf->S, &kf->temp_matrix1); // 计算inv(H·P'(k)·HT + R)
kf->temp_matrix.numRows = kf->Pminus.numRows; // 设置临时矩阵的行数
kf->temp_matrix.numCols = kf->HT.numCols; // 设置临时矩阵的列数
kf->MatStatus = Matrix_Multiply(&kf->Pminus, &kf->HT, &kf->temp_matrix); // 计算P'(k)·HT
kf->MatStatus = Matrix_Multiply(&kf->temp_matrix, &kf->temp_matrix1, &kf->K); // 计算K(k)
}
}
void Kalman_Filter_xhatUpdate(KalmanFilter_t *kf)
{
if (!kf->SkipEq4) // 如果未跳过此步骤
{
kf->temp_vector.numRows = kf->H.numRows; // 设置临时矩阵的行数
kf->temp_vector.numCols = 1; // 设置临时矩阵的列数
kf->MatStatus = Matrix_Multiply(&kf->H, &kf->xhatminus, &kf->temp_vector); // 计算H·xhat'(k)
kf->temp_vector1.numRows = kf->z.numRows; // 设置另一个临时矩阵的行数
kf->temp_vector1.numCols = 1; // 设置另一个临时矩阵的列数
kf->MatStatus = Matrix_Subtract(&kf->z, &kf->temp_vector, &kf->temp_vector1); // 计算z(k) - H·xhat'(k)
kf->temp_vector.numRows = kf->K.numRows; // 设置临时矩阵的行数
kf->temp_vector.numCols = 1; // 设置临时矩阵的列数
kf->MatStatus = Matrix_Multiply(&kf->K, &kf->temp_vector1, &kf->temp_vector); // 计算K(k)·(z(k) - H·xhat'(k))
kf->MatStatus = Matrix_Add(&kf->xhatminus, &kf->temp_vector, &kf->xhat); // 计算xhat(k)
}
}
void Kalman_Filter_P_Update(KalmanFilter_t *kf)
{

{
kf->temp_matrix.numRows = kf->K.numRows; // 设置临时矩阵的行数

kf->MatStatus = Matrix_Multiply(&kf->K, &kf->H, &kf->temp_matrix); // 计算K(k)·H

kf->temp_matrix1.numCols = kf->Pminus.numCols; // 设置另一个临时矩阵的列数

float *Kalman_Filter_Update(KalmanFilter_t *kf)
{
// 0. 获取量测信息
Kalman_Filter_Measure(kf);
if (kf->User_Func0_f != NULL)
kf->User_Func0_f(kf);
// 先验估计
// 1. xhat'(k)= A·xhat(k-1) + B·u
Kalman_Filter_xhatMinusUpdate(kf);
if (kf->User_Func1_f != NULL)
kf->User_Func1_f(kf);
// 预测更新
// 2. P'(k) = A·P(k-1)·AT + Q
Kalman_Filter_PminusUpdate(kf);
if (kf->User_Func2_f != NULL)
kf->User_Func2_f(kf);
if (kf->MeasurementValidNum != 0 || kf->UseAutoAdjustment == 0)
{
// 量测更新
// 3. K(k) = P'(k)·HT / (H·P'(k)·HT + R)
Kalman_Filter_SetK(kf);
if (kf->User_Func3_f != NULL)
kf->User_Func3_f(kf);
// 融合
// 4. xhat(k) = xhat'(k) + K(k)·(z(k) - H·xhat'(k))
Kalman_Filter_xhatUpdate(kf);
if (kf->User_Func4_f != NULL)
kf->User_Func4_f(kf);
// 修正方差
// 5. P(k) = (1-K(k)·H)·P'(k) ==> P(k) = P'(k)-K(k)·H·P'(k)
Kalman_Filter_P_Update(kf);
}
else
{
// 无有效量测,仅预测
// xhat(k) = xhat'(k)
// P(k) = P'(k)
memcpy(kf->xhat_data, kf->xhatminus_data, sizeof_float * kf->xhatSize);
memcpy(kf->P_data, kf->Pminus_data, sizeof_float * kf->xhatSize * kf->xhatSize);
}
// 自定义函数,可以提供后处理等
if (kf->User_Func5_f != NULL)
kf->User_Func5_f(kf);
// 避免滤波器过度收敛
// suppress filter excessive convergence
for (uint8_t i = 0; i < kf->xhatSize; i++)
{
if (kf->P_data[i * kf->xhatSize + i] < kf->StateMinVariance[i])
kf->P_data[i * kf->xhatSize + i] = kf->StateMinVariance[i];
}
memcpy(kf->FilteredValue, kf->xhat_data, sizeof_float * kf->xhatSize);
if (kf->User_Func6_f != NULL)
kf->User_Func6_f(kf);
return kf->FilteredValue;
}
static void H_K_R_Adjustment(KalmanFilter_t *kf)
{
kf->MeasurementValidNum = 0;
memcpy(kf->z_data, kf->MeasuredVector, sizeof_float * kf->zSize);
memset(kf->MeasuredVector, 0, sizeof_float * kf->zSize);
// 识别量测数据有效性并调整矩阵H R K
// recognize measurement validity and adjust matrices H R K
memset(kf->R_data, 0, sizeof_float * kf->zSize * kf->zSize);
memset(kf->H_data, 0, sizeof_float * kf->xhatSize * kf->zSize);
for (uint8_t i = 0; i < kf->zSize; i++)
{
if (kf->z_data[i] != 0)
{
// 重构向量z
// rebuild vector z
kf->z_data[kf->MeasurementValidNum] = kf->z_data[i];
kf->temp[kf->MeasurementValidNum] = i;
// 重构矩阵H
// rebuild matrix H
kf->H_data[kf->xhatSize * kf->MeasurementValidNum + kf->MeasurementMap[i] - 1] = kf->MeasurementDegree[i];
kf->MeasurementValidNum++;
}
}
for (uint8_t i = 0; i < kf->MeasurementValidNum; i++)
{
// 重构矩阵R
// rebuild matrix R
kf->R_data[i * kf->MeasurementValidNum + i] = kf->MatR_DiagonalElements[kf->temp[i]];
}
// 调整矩阵维数
// adjust the dimensions of system matrices
kf->H.numRows = kf->MeasurementValidNum;
kf->H.numCols = kf->xhatSize;
kf->HT.numRows = kf->xhatSize;
kf->HT.numCols = kf->MeasurementValidNum;
kf->R.numRows = kf->MeasurementValidNum;
kf->R.numCols = kf->MeasurementValidNum;
kf->K.numRows = kf->xhatSize;
kf->K.numCols = kf->MeasurementValidNum;
kf->z.numRows = kf->MeasurementValidNum;
}
